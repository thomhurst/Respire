using BenchmarkDotNet.Attributes;
using BenchmarkDotNet.Running;
using Respire;
using Respire.Internal;
using Respire.TimeSeries;
using Respire.Networking;

using var output = Environment.GetEnvironmentVariable("RESPIRE_944_OUTPUT") is { } path
    ? new StreamWriter(path) { AutoFlush = true } : null;
if (output is not null) Console.SetOut(output);

if (args.Contains("--fairness")) await Fairness.RunAsync();
else BenchmarkSwitcher.FromAssembly(typeof(FollowerLeases).Assembly).Run(args);

[MemoryDiagnoser]
public class FollowerLeases
{
    public IEnumerable<int> PageSizes => Environment.GetEnvironmentVariable("RESPIRE_944_PAGE") is { } value
        ? [int.Parse(value)] : [1, 8, 64];
    public IEnumerable<int> ConsumerDelays => Environment.GetEnvironmentVariable("RESPIRE_944_DELAY") is { } value
        ? [int.Parse(value)] : [0, 1];
    [ParamsSource(nameof(PageSizes))] public int PageSize { get; set; }
    [ParamsSource(nameof(ConsumerDelays))] public int ConsumerDelayMilliseconds { get; set; }
    private RespireClient _client = null!;
    private RespireTimeSeriesClient _series = null!;
    private string _key = null!;

    internal static ValueTask<RespireClient> ConnectAsync() => RespireClient.ConnectAsync(new RespireOptions
    {
        Endpoints = [new("127.0.0.1", int.Parse(Environment.GetEnvironmentVariable("RESPIRE_944_PORT")!))],
        Connections = 1,
        CommandTimeout = TimeSpan.FromSeconds(5),
        MaintenanceNotifications = RespireMaintenanceNotificationMode.Disabled,
    });

    [GlobalSetup]
    public async Task Setup()
    {
        _client = await ConnectAsync();
        _series = new(_client);
        _key = "issue944:" + Guid.NewGuid().ToString("N");
        for (var i = 1; i <= 64; i++) await _series.AddAsync(_key, i, i);
        // Establish the dedicated socket before measurement.
        await _series.ReadAsync(_key, 1, new() { BlockMilliseconds = 0, MaximumCount = 1 });
    }

    [Benchmark(Baseline = true)] public Task<double> PerPage() => ConsumeAsync(false);
    [Benchmark] public Task<double> PersistentPrototype() => ConsumeAsync(true);

    private async Task<double> ConsumeAsync(bool retain)
    {
        using var scope = retain ? new Issue944LeaseScope() : null;
        using var cancellation = new CancellationTokenSource(TimeSpan.FromSeconds(10));
        var count = 0;
        double sum = 0;
        // One operation is a complete 64-sample consumption workload, not a microbenchmark loop.
        await foreach (var sample in _series.FollowAsync(_key, 1, PageSize, cancellation.Token))
        {
            sum += sample.Value;
            if (ConsumerDelayMilliseconds != 0) await Task.Delay(ConsumerDelayMilliseconds);
            if (++count == 64) break;
        }
        if (sum != 2080) throw new InvalidOperationException("Unexpected sample sequence.");
        return sum;
    }

    [GlobalCleanup]
    public async Task Cleanup()
    {
        using (await _client.ExecuteAsync("DEL", [_key])) { }
        await _client.DisposeAsync();
    }
}

[MemoryDiagnoser]
public class PoolBookkeeping
{
    private RespireClient _client = null!;
    private DedicatedConnectionPool _pool = null!;
    private RespireConnection _held = null!;

    [GlobalSetup]
    public async Task Setup()
    {
        _client = await FollowerLeases.ConnectAsync();
        _pool = await _client.Core.GetDedicatedPoolAsync(CancellationToken.None);
        _held = await _pool.RentAsync(CancellationToken.None);
        var idle = await _pool.RentAsync(CancellationToken.None);
        _pool.Return(idle);
    }

    [Benchmark(Baseline = true)]
    public async ValueTask<bool> PerPageRentReturn()
    {
        var connection = await _pool.RentAsync(CancellationToken.None);
        var healthy = connection.IsConnected;
        _pool.Return(connection);
        return healthy;
    }

    // Isolates lease bookkeeping only; this is not a replacement for route/error handling.
    [Benchmark]
    public bool RetainedLeaseCheck() => !_pool.IsStopping && _pool.IsMovingPublicationCurrent && _held.IsConnected;

    [GlobalCleanup]
    public async Task Cleanup()
    {
        _pool.Return(_held);
        await _client.DisposeAsync();
    }
}

internal static class Fairness
{
    internal static async Task RunAsync()
    {
        foreach (var retain in new[] { false, true })
        {
            await using var client = await FollowerLeases.ConnectAsync();
            var series = new RespireTimeSeriesClient(client);
            var key = "issue944:fair:" + Guid.NewGuid().ToString("N");
            await series.AddAsync(key, 1, 1);
            var pool = await client.Core.GetDedicatedPoolAsync(CancellationToken.None);
            var resume = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
            var ready = Enumerable.Range(0, 8).Select(_ => new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously)).ToArray();
            async Task PauseFollower(TaskCompletionSource signal)
            {
                using var scope = retain ? new Issue944LeaseScope() : null;
                await using var reader = series.FollowAsync(key, 1, 1).GetAsyncEnumerator();
                if (!await reader.MoveNextAsync()) throw new InvalidOperationException();
                signal.SetResult();
                await resume.Task;
            }
            var followers = new List<Task>();
            foreach (var signal in ready)
            {
                followers.Add(PauseFollower(signal));
                await signal.Task.WaitAsync(TimeSpan.FromSeconds(10));
            }
            var paused = pool.CaptureRetirementState().Borrowed;
            using var before = await client.ExecuteAsync("CLIENT LIST", []);
            var beforeCount = before.AsString().Split('\n', StringSplitOptions.RemoveEmptyEntries).Length;
            // This call is outside every follower scope and must get an independent lease.
            using var cancellation = new CancellationTokenSource(TimeSpan.FromSeconds(5));
            var timer = System.Diagnostics.Stopwatch.StartNew();
            var competitor = await series.ReadAsync(key, 1, new() { BlockMilliseconds = 0, MaximumCount = 1 }, cancellation.Token);
            timer.Stop();
            using var after = await client.ExecuteAsync("CLIENT LIST", []);
            var afterCount = after.AsString().Split('\n', StringSplitOptions.RemoveEmptyEntries).Length;
            var retirement = pool.RetireAsync().AsTask();
            await Task.WhenAny(retirement, Task.Delay(100));
            var retiredWhilePaused = retirement.IsCompletedSuccessfully;
            resume.SetResult();
            await Task.WhenAll(followers).WaitAsync(TimeSpan.FromSeconds(10));
            await retirement.WaitAsync(TimeSpan.FromSeconds(10));
            var released = pool.CaptureRetirementState().Borrowed;
            Console.WriteLine($"retain={retain}; followers=8; pausedBorrowed={paused}; clientsBefore={beforeCount}; clientsAfter={afterCount}; competitorSamples={competitor.Samples.Count}; competitorMs={timer.Elapsed.TotalMilliseconds:F3}; retiredWhilePaused={retiredWhilePaused}; afterDisposeBorrowed={released}");
            if (released != 0 || competitor.Samples.Count != 1) throw new InvalidOperationException("Cleanup/fairness failed.");
            using (await client.ExecuteAsync("DEL", [key])) { }
            await using var cancelClient = await FollowerLeases.ConnectAsync();
            var cancelPool = await cancelClient.Core.GetDedicatedPoolAsync(CancellationToken.None);
            var cancelSeries = new RespireTimeSeriesClient(cancelClient);
            await cancelSeries.AddAsync(key, 1, 1);
            using (var scope = retain ? new Issue944LeaseScope() : null)
            using (var cancel = new CancellationTokenSource(TimeSpan.FromMilliseconds(100)))
            {
                await using var reader = cancelSeries.FollowAsync(key, 2, 1, cancel.Token).GetAsyncEnumerator();
                try { await reader.MoveNextAsync(); throw new InvalidOperationException("Expected cancellation."); }
                catch (OperationCanceledException) when (cancel.IsCancellationRequested) { }
            }
            Console.WriteLine($"retain={retain}; canceledBorrowed={cancelPool.CaptureRetirementState().Borrowed}");
            if (cancelPool.CaptureRetirementState().Borrowed != 0) throw new InvalidOperationException("Cancellation leaked lease.");
            using (await cancelClient.ExecuteAsync("DEL", [key])) { }
        }
    }
}
