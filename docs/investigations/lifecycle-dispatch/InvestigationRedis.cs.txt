using System.Globalization;
using DotNet.Testcontainers.Configurations;
using Testcontainers.Redis;

namespace Respire.Benchmarks;

internal static class InvestigationRedis
{
    internal static RedisContainer Build()
    {
        if (Environment.GetEnvironmentVariable("RESP_LIFECYCLE_REAPER_PORT") is { } reaperValue)
        {
            var reaperPort = ushort.Parse(reaperValue, CultureInfo.InvariantCulture);
            // This task-only hook is read by the pinned Testcontainers 4.15.0 reaper constructor.
#pragma warning disable CS0618
            TestcontainersSettings.ResourceReaperPublicHostPort = _ => reaperPort;
#pragma warning restore CS0618
        }
        var builder = new RedisBuilder("redis:7.2-alpine");
        if (Environment.GetEnvironmentVariable("RESP_LIFECYCLE_REDIS_PORT") is { } value)
            builder = builder.WithPortBinding(int.Parse(value, CultureInfo.InvariantCulture), 6379);
        return builder.Build();
    }
}
