param(
    [Parameter(Mandatory)][ValidateSet('net8.0', 'net10.0')][string]$Framework,
    [Parameter(Mandatory)][string]$Evidence,
    [string[]]$Cases = @(),
    [string]$RunTag = '',
    [int]$RedisPort = 26363,
    [int]$ReaperPortBase = 26500
)
$ErrorActionPreference = 'Stop'
$worktree = (Resolve-Path (Join-Path $PSScriptRoot '../..')).Path
$evidence = $Evidence
$guard = Join-Path $worktree 'scripts/Invoke-AgentDotNet.ps1'
$env:RESP_LIFECYCLE_REDIS_PORT = [string]$RedisPort
& $guard -SingleNode -DotNetArguments @('build', 'artifacts/issue-1063-investigation/Respire.Benchmarks.csproj', '-c', 'Release', '-f', $Framework)
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
$dll = Join-Path $worktree "artifacts/issue-1063-investigation/bin/Release/$Framework/Respire.Benchmarks.dll"
foreach ($method in @('WarmIdleRental', 'NewConnectionRental')) {
    foreach ($mode in @('Disabled', 'Fast', 'Slow')) {
        if ($Cases.Count -ne 0 -and "$method-$mode" -notin $Cases) { continue }
        $case = "$method-$mode-$Framework"
        $runtimeOffset = if ($Framework -eq 'net8.0') { 0 } else { 10 }
        $methodOffset = if ($method -eq 'WarmIdleRental') { 0 } else { 3 }
        $modeOffset = @('Disabled', 'Fast', 'Slow').IndexOf($mode)
        $env:RESP_LIFECYCLE_REAPER_PORT = [string]($ReaperPortBase + $runtimeOffset + $methodOffset + $modeOffset)
        if (@(Get-NetTCPConnection -LocalPort $RedisPort,([int]$env:RESP_LIFECYCLE_REAPER_PORT) -ErrorAction SilentlyContinue).Count -ne 0) {
            throw "Investigation Redis or reaper port is occupied: $case"
        }
        $output = Join-Path $evidence "isolated-$RunTag$case"
        New-Item -ItemType Directory -Path $output | Out-Null
        $env:RESP_LIFECYCLE_DELIVERY_LOG = Join-Path $output 'delivery.jsonl'
        & $guard -SingleNode -DotNetArguments @($dll, '--filter', "*LifecycleBenchmarks.$method*$mode*", '--inProcess', '--artifacts', $output, '--noOverwrite', '--exporters', 'json')
        if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
        $reports = @(Get-ChildItem $output -Recurse -Filter '*report-full-compressed.json')
        if ($reports.Count -ne 1) { throw "Expected one report: $case" }
        $data = Get-Content $reports[0].FullName -Raw | ConvertFrom-Json
        if (@($data.Benchmarks).Count -ne 1 -or $null -eq $data.Benchmarks[0].Statistics) { throw "Incomplete case: $case" }
        $delivery = @(Get-Content $env:RESP_LIFECYCLE_DELIVERY_LOG | ForEach-Object { $_ | ConvertFrom-Json })
        if ($delivery.Count -ne 1 -or $delivery[0].Pending -ne 0) { throw "Incomplete drain: $case" }
        if ($method -eq 'WarmIdleRental') {
            $expected = if ($mode -eq 'Disabled') { 0 } else { 1 }
            if ($delivery[0].Delivered -ne $expected -or $delivery[0].Dropped -ne 0) { throw "Warm rental emitted unexpected events: $case" }
        }
        [pscustomobject]@{Case=$case;Mean=$data.Benchmarks[0].Statistics.Mean;Allocated=$data.Benchmarks[0].Memory.BytesAllocatedPerOperation;Delivered=$delivery[0].Delivered;Dropped=$delivery[0].Dropped} | ConvertTo-Json -Compress
    }
}
