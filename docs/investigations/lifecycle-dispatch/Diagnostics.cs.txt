using System.Diagnostics;
using System.Diagnostics.Metrics;
using System.Text.Json;
using Respire;
using Respire.Internal;
using Respire.Networking;
using Testcontainers.Redis;

namespace Respire.Benchmarks;

internal static class Diagnostics
{
    internal static async Task RunAsync()
    {
        RespireMetrics.Configure(new() { Groups = RespireMetricGroups.All });
        await using var redis = InvestigationRedis.Build();
        await redis.StartAsync();
        await using var pool = new DedicatedConnectionPool(redis.Hostname, redis.GetMappedPublicPort(6379),
            new() { Protocol = RespProtocol.Resp2 }, null);
        var connection = await pool.RentAsync(default);
        using var entered = new ManualResetEventSlim(false);
        using var release = new ManualResetEventSlim(false);
        using var listener = new MeterListener();
        long delivered = 0;
        listener.InstrumentPublished = (instrument, current) =>
        {
            if (instrument.Meter.Name == "Respire" && instrument.Name == "db.client.connection.wait_time")
                current.EnableMeasurementEvents(instrument);
        };
        listener.SetMeasurementEventCallback<double>((_, _, _, _) =>
        {
            Interlocked.Increment(ref delivered);
            entered.Set();
            release.Wait();
        });
        listener.Start();
        var droppedBefore = ConnectionTelemetry.DroppedMeasurements;
        try
        {
            connection.RecordConnectionWait(Stopwatch.GetTimestamp());
            if (!entered.Wait(TimeSpan.FromSeconds(5))) throw new TimeoutException("Listener did not enter.");
            var checkpoints = new List<object>();
            var watch = Stopwatch.StartNew();
            var offered = 1;
            foreach (var target in new[] { 64, 1_000, 100_000, 1_000_000 })
            {
                while (offered < target)
                {
                    connection.RecordConnectionWait(Stopwatch.GetTimestamp());
                    offered++;
                }
                checkpoints.Add(new
                {
                    Offered = offered, Pending = ConnectionTelemetry.PendingMeasurements,
                    Dropped = ConnectionTelemetry.DroppedMeasurements - droppedBefore,
                    LiveManagedBytes = GC.GetTotalMemory(true),
                    WorkingSetBytes = Process.GetCurrentProcess().WorkingSet64,
                    ElapsedMilliseconds = watch.Elapsed.TotalMilliseconds
                });
            }
            RespireMetrics.Configure(new() { Groups = RespireMetricGroups.None });
            var disposal = Stopwatch.StartNew();
            await pool.DisposeAsync();
            disposal.Stop();
            Console.WriteLine("SATURATION " + JsonSerializer.Serialize(new
            {
                Runtime = System.Runtime.InteropServices.RuntimeInformation.FrameworkDescription,
                Checkpoints = checkpoints, DisposalMilliseconds = disposal.Elapsed.TotalMilliseconds,
                ConnectedAfterDisposal = connection.IsConnected
            }));
        }
        finally
        {
            release.Set();
            using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(30));
            while (ConnectionTelemetry.PendingMeasurements != 0) await Task.Delay(1, deadline.Token);
        }
        Console.WriteLine("DRAIN " + JsonSerializer.Serialize(new
        {
            Delivered = Interlocked.Read(ref delivered), Pending = ConnectionTelemetry.PendingMeasurements,
            Dropped = ConnectionTelemetry.DroppedMeasurements - droppedBefore
        }));
        listener.Dispose();
        await AcquisitionLatencyAsync(redis);
        Isolation();
    }

    private static async Task AcquisitionLatencyAsync(RedisContainer redis)
    {
        foreach (var mode in Enum.GetValues<ListenerMode>())
        {
            RespireMetrics.Configure(new() { Groups = RespireMetricGroups.All });
            await using var pool = new DedicatedConnectionPool(redis.Hostname, redis.GetMappedPublicPort(6379),
                new() { Protocol = RespProtocol.Resp2 }, null);
            var initial = await pool.RentAsync(default);
            pool.Return(initial);
            long delivered = 0;
            using var listener = new MeterListener();
            listener.InstrumentPublished = (instrument, current) =>
            {
                if (mode != ListenerMode.Disabled && instrument.Meter.Name == "Respire" && instrument.Name is
                    "db.client.connection.create_time" or "db.client.connection.wait_time" or "redis.client.connection.closed")
                    current.EnableMeasurementEvents(instrument);
            };
            void Delivered()
            {
                if (mode == ListenerMode.Slow) Thread.Sleep(1);
                Interlocked.Increment(ref delivered);
            }
            listener.SetMeasurementEventCallback<double>((_, _, _, _) => Delivered());
            listener.SetMeasurementEventCallback<long>((_, _, _, _) => Delivered());
            listener.Start();
            for (var i = 0; i < 20; i++)
            {
                var warm = await pool.RentAsync(default, reuseIdle: false);
                await pool.DiscardAsync(warm);
            }
            await DrainAsync();
            var latencies = new double[1000];
            var deliveredBefore = Interlocked.Read(ref delivered);
            var droppedBefore = ConnectionTelemetry.DroppedMeasurements;
            var allocatedBefore = GC.GetTotalAllocatedBytes(true);
            var elapsed = Stopwatch.StartNew();
            for (var i = 0; i < latencies.Length; i++)
            {
                var started = Stopwatch.GetTimestamp();
                var rented = await pool.RentAsync(default, reuseIdle: false);
                latencies[i] = Stopwatch.GetElapsedTime(started).TotalMicroseconds;
                if (!rented.IsConnected) throw new InvalidOperationException("Disconnected rental.");
                await pool.DiscardAsync(rented);
            }
            await DrainAsync();
            elapsed.Stop();
            var allocated = GC.GetTotalAllocatedBytes(true) - allocatedBefore;
            Array.Sort(latencies);
            var completed = Interlocked.Read(ref delivered) - deliveredBefore;
            Console.WriteLine("ACQUISITION " + JsonSerializer.Serialize(new
            {
                Mode = mode.ToString(), Rentals = latencies.Length,
                MedianMicroseconds = latencies[500], P95Microseconds = latencies[950],
                MeanMicroseconds = latencies.Average(),
                ProcessAllocatedBytesPerRental = allocated / (double)latencies.Length,
                Delivered = completed, Dropped = ConnectionTelemetry.DroppedMeasurements - droppedBefore,
                DeliveredPerSecond = completed / elapsed.Elapsed.TotalSeconds,
                ElapsedMilliseconds = elapsed.Elapsed.TotalMilliseconds
            }));
        }
    }

    private static async Task DrainAsync()
    {
        using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(30));
        while (ConnectionTelemetry.PendingMeasurements != 0) await Task.Delay(1, deadline.Token);
    }

    private static void Isolation()
    {
        foreach (var readers in new[] { 0, 1, 4, 64 })
        {
            Dispatch dispatch = readers == 0 ? new WorkItemDispatch() : new ChannelDispatch(readers);
            using var meter = new Meter("IsolationInvestigation");
            var histogram = meter.CreateHistogram<double>("wait");
            using var entered = new ManualResetEventSlim(false);
            using var released = new ManualResetEventSlim(false);
            using var available = new ManualResetEventSlim(false);
            using var listener = new MeterListener();
            var blocked = readers == 1 ? 1 : 4;
            var enteredCount = 0;
            listener.InstrumentPublished = (instrument, current) =>
            {
                if (instrument.Meter == meter) current.EnableMeasurementEvents(instrument);
            };
            listener.SetMeasurementEventCallback<double>((_, _, tags, _) =>
            {
                if (Equals(tags[0].Value, "available")) { available.Set(); return; }
                if (Interlocked.Increment(ref enteredCount) == blocked) entered.Set();
                released.Wait();
            });
            listener.Start();
            try
            {
                for (var i = 0; i < blocked; i++)
                    if (!dispatch.Offer(new(dispatch, histogram, [new("pool", "blocked")], 0.001)))
                        throw new InvalidOperationException("Blocked sample rejected below capacity.");
                if (!entered.Wait(TimeSpan.FromSeconds(10))) throw new TimeoutException("Blocked samples did not enter.");
                if (!dispatch.Offer(new(dispatch, histogram, [new("pool", "available")], 0.001)))
                    throw new InvalidOperationException("Available sample rejected below capacity.");
                var beforeRelease = available.Wait(TimeSpan.FromMilliseconds(100));
                Console.WriteLine("ISOLATION " + JsonSerializer.Serialize(new
                {
                    Readers = readers, BlockedCallbacks = blocked,
                    UnrelatedDeliveredBeforeRelease = beforeRelease, Dropped = dispatch.Dropped
                }));
                if ((readers is 0 or 64) != beforeRelease) throw new InvalidOperationException("Unexpected isolation result.");
            }
            finally
            {
                released.Set();
                dispatch.WaitForDrain();
                if (dispatch is ChannelDispatch channel) channel.Dispose();
            }
        }
    }
}
